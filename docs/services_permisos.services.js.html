/**
 * @module permisosService
 * @description
 * Service encargado de manejar la lógica de negocio de los permisos
 * dentro de la arquitectura MVC.
 *
 * Responsabilidades:
 * - Aplicar reglas de negocio
 * - Validar datos antes de enviarlos al repositorio
 * - Comunicarse con el Repository para acceder a los datos
 *
 * Flujo:
 * Controller → Service → Repository → Base de Datos
 */

// Importa el repository encargado del acceso a datos
const permisoRepository = require('../repositories/permisos.repository');

/**
 * Crear un nuevo permiso
 *
 * Regla de negocio:
 * No se puede crear un permiso con un nombre que ya exista.
 *
 * @function createPermiso
 * @memberof module:permisosService
 *
 * @param {Object} data - Datos del permiso a crear
 * @param {string} data.nombre_permiso - Nombre del permiso
 * @returns {Object} Permiso creado
 * @throws {Error} Si el permiso ya existe
 */
const createPermiso = (data) => {

  const existing = permisoRepository.getByName(data.nombre_permiso);

  if (existing) {
    throw new Error('El permiso ya existe');
  }

  return permisoRepository.create(data);
};

/**
 * Obtener todos los permisos
 *
 * @function getPermisos
 * @memberof module:permisosService
 *
 * @returns {Array<Object>} Lista de permisos
 */
const getPermisos = () => permisoRepository.getAll();

/**
 * Obtener un permiso por id
 *
 * @function getPermisosById
 * @memberof module:permisosService
 *
 * @returns {Array<Object>} Lista de permisos
 */
const getPermisosById = (id) => permisoRepository.getById(id);


/**
 * Editar un permiso existente
 *
 * Reglas de negocio:
 * - El permiso debe existir
 * - El nombre del permiso no puede duplicarse
 *
 * @function editPermiso
 * @memberof module:permisosService
 *
 * @param {number} id - ID del permiso a editar
 * @param {Object} data - Nuevos datos del permiso
 * @param {string} data.nombre_permiso - Nuevo nombre del permiso
 *
 * @returns {Object} Permiso actualizado
 * @throws {Error} Si el permiso no existe
 * @throws {Error} Si el nombre del permiso ya existe
 */
const editPermiso = (id, data) => {

  const permisoExisting = permisoRepository.getById(id);

  if (!permisoExisting) {
    throw new Error("El permiso no existe");
  }

  const duplicatePermiso = permisoRepository.getByName(data.nombre_permiso);

  if (duplicatePermiso && duplicatePermiso.id_permiso !== id) {
    throw new Error("Ya existe un permiso con ese nombre");
  }

  return permisoRepository.editById(id, data);
};


/**
 * Eliminar un permiso por su ID
 *
 * Regla de negocio:
 * - El permiso debe existir antes de eliminarse
 *
 * @function deletePermisoById
 * @memberof module:permisosService
 *
 * @param {number} id - ID del permiso a eliminar
 * @returns {Object} Permiso eliminado
 * @throws {Error} Si el permiso no existe
 */
const deletePermisoById = (id) => {

  const permisoExisting = permisoRepository.getById(id);

  if (!permisoExisting) {
    throw new Error("El permiso no existe");
  }

  return permisoRepository.deleteById(id);
};


module.exports = {
  createPermiso,
  getPermisos,
  getPermisosById,
  editPermiso,
  deletePermisoById
};
